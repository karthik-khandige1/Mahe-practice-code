#include <stdio.h>
#include <string.h>
int main() {
    char num[50];
    printf("Enter a number: ");
    scanf("%s", num);
    int start = 0;
    int end = strlen(num) - 1;
    while (start < end) {
        if (num[start] != num[end]) {
            printf("%s is NOT a palindrome.\n", num);
            return 0;
        }
        start++;
        end--;
    }
    printf("%s is a palindrome.\n", num);
    return 0;
}
